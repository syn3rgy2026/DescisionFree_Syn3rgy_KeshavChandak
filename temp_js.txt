
const status = document.getElementById('status');
const cells = document.querySelectorAll('.cell');
const reset = document.getElementById('reset');
let player = 'X', state = Array(9).fill(''), active = true;
const wins = [[0,1,2],[3,4,5],[6,7,8],[0,3,6],[1,4,7],[2,5,8],[0,4,8],[2,4,6]];
cells.forEach(c => c.addEventListener('click', e => {
    const i = e.target.dataset.index;
    if(!state[i] && active) {
        state[i] = player; e.target.innerText = player;
        if(wins.some(w => w.every(j => state[j] === player))) {
            status.innerText = `Player ${player} Wins!`; active = false;
        } else if(!state.includes('')) {
            status.innerText = "Draw!"; active = false;
        } else {
            player = player === 'X' ? 'O' : 'X';
            status.innerText = `Player ${player}'s Turn`;
        }
    }
}));
reset.addEventListener('click', () => {
    player = 'X'; state.fill(''); active = true;
    status.innerText = "Player X's Turn";
    cells.forEach(c => c.innerText = '');
});
